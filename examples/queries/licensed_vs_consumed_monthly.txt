-- Licensed vs. consumed, per product, per month, for one account (last 12 months).
-- Fill in your own database/schema/table (a usage-vs-entitlement fact table)
-- and swap the ACCOUNT_NAME filter for a real customer.
SELECT
  ACCOUNT_NAME,
  METRIC_MONTH,
  PRODUCT_FAMILY,
  BILLABLE_METRIC_NAME,
  CONTRACTED_QUANTITY,
  MONTHLY_VALUE,
  UTILIZATION_RATE_LIFETIME
FROM YOUR_DB.YOUR_SCHEMA.YOUR_USAGE_TABLE
WHERE ACCOUNT_NAME = 'YOUR_ACCOUNT_NAME'
ORDER BY METRIC_MONTH DESC
LIMIT 12
