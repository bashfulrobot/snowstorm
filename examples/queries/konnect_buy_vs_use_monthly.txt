-- Licensed vs. consumed, per product, per month, for one account (last 12 months).
-- Swap the ACCOUNT_NAME filter for a real customer.
SELECT
  ACCOUNT_NAME,
  METRIC_MONTH,
  PRODUCT_FAMILY,
  BILLABLE_METRIC_NAME,
  CONTRACTED_QUANTITY,
  MONTHLY_VALUE,
  UTILIZATION_RATE_LIFETIME
FROM REVANALYTICS.REPORTING.KONNECT_BUY_VS_USE_MONTHLY
WHERE ACCOUNT_NAME = '9fin'
ORDER BY METRIC_MONTH DESC
LIMIT 12
